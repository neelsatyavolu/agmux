# IDE Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a global IDE mode to Xanom with a Cursor/Windsurf-style layout — file tree (left), code editor (center), integrated terminal (bottom), and SDK chat sidebar (right).

**Architecture:** New `IdeLayout` top-level component replaces Sidebar + MainPanel when `uiStore.appMode === "ide"`. Reuses existing components (FileTree, CodeEditor, ClaudeSdkSessionView, TerminalView, ClaudeInputBar) composed in a fixed four-zone layout with resizable panels. New `ideStore` manages IDE-specific persisted state (panel sizes, terminal sessions, active chat thread per project). New Rust command `get_git_status` provides git status for the enhanced file tree.

**Tech Stack:** React 19, TypeScript 5.8, Zustand 5, Tailwind CSS v4, Framer Motion 12, ghostty-web (WASM), Tauri v2, Rust (sqlx, nix)

**Spec:** `docs/superpowers/specs/2026-04-05-ide-mode-design.md`

```typescript
// src/stores/ideStore.ts
import { create } from "zustand";
import { persist } from "zustand/middleware";

const MAX_TERMINAL_TABS = 4;

const EMPTY_STRING_ARRAY: string[] = [];

interface IdeState {
  // Panel dimensions
  fileTreeWidth: number;
  chatPanelWidth: number;
  terminalHeight: number; // ratio 0–1
  fileTreeCollapsed: boolean;
  chatPanelCollapsed: boolean;

  // Terminal state
  terminalVisible: boolean;
  terminalSessions: Record<string, string[]>; // projectId -> PTY session IDs
  terminalActiveTab: Record<string, number>; // projectId -> active tab index

  // Chat state
  activeIdeChatThreadId: Record<string, string | null>; // projectId -> thread ID

  // Actions
  setFileTreeWidth: (width: number) => void;
  setChatPanelWidth: (width: number) => void;
  setTerminalHeight: (height: number) => void;
  toggleFileTree: () => void;
  toggleChatPanel: () => void;
  toggleTerminal: () => void;
  addTerminalSession: (projectId: string, sessionId: string) => void;
  removeTerminalSession: (projectId: string, sessionId: string) => void;
  setTerminalActiveTab: (projectId: string, index: number) => void;
  setActiveIdeChatThread: (projectId: string, threadId: string | null) => void;
  getTerminalSessions: (projectId: string) => string[];
  getTerminalActiveTab: (projectId: string) => number;
  getActiveIdeChatThreadId: (projectId: string) => string | null;
}

export const useIdeStore = create<IdeState>()(
  persist(
    (set, get) => ({
      // Defaults
      fileTreeWidth: 240,
      chatPanelWidth: 380,
      terminalHeight: 0.3,
      fileTreeCollapsed: false,
      chatPanelCollapsed: false,
      terminalVisible: false,
      terminalSessions: {},
      terminalActiveTab: {},
      activeIdeChatThreadId: {},

      // Panel dimension actions
      setFileTreeWidth: (width) => set({ fileTreeWidth: Math.max(180, Math.min(500, width)) }),
      setChatPanelWidth: (width) => set({ chatPanelWidth: Math.max(280, Math.min(600, width)) }),
      setTerminalHeight: (height) => set({ terminalHeight: Math.max(0.1, Math.min(0.8, height)) }),

      // Toggle actions
      toggleFileTree: () => set((s) => ({ fileTreeCollapsed: !s.fileTreeCollapsed })),
      toggleChatPanel: () => set((s) => ({ chatPanelCollapsed: !s.chatPanelCollapsed })),
      toggleTerminal: () => set((s) => ({ terminalVisible: !s.terminalVisible })),

      // Terminal session actions
      addTerminalSession: (projectId, sessionId) =>
        set((s) => {
          const current = s.terminalSessions[projectId] ?? EMPTY_STRING_ARRAY;
          if (current.length >= MAX_TERMINAL_TABS) return s;
          if (current.includes(sessionId)) return s;
          return {
            terminalSessions: {
              ...s.terminalSessions,
              [projectId]: [...current, sessionId],
            },
            terminalActiveTab: {
              ...s.terminalActiveTab,
              [projectId]: current.length, // focus new tab
            },
          };
        }),

      removeTerminalSession: (projectId, sessionId) =>
        set((s) => {
          const current = s.terminalSessions[projectId] ?? EMPTY_STRING_ARRAY;
          const filtered = current.filter((id) => id !== sessionId);
          const activeIdx = s.terminalActiveTab[projectId] ?? 0;
          return {
            terminalSessions: {
              ...s.terminalSessions,
              [projectId]: filtered,
            },
            terminalActiveTab: {
              ...s.terminalActiveTab,
              [projectId]: Math.min(activeIdx, Math.max(0, filtered.length - 1)),
            },
          };
        }),

      setTerminalActiveTab: (projectId, index) =>
        set((s) => ({
          terminalActiveTab: { ...s.terminalActiveTab, [projectId]: index },
        })),

      setActiveIdeChatThread: (projectId, threadId) =>
        set((s) => ({
          activeIdeChatThreadId: { ...s.activeIdeChatThreadId, [projectId]: threadId },
        })),

      // Getters (stable empty returns)
      getTerminalSessions: (projectId) => get().terminalSessions[projectId] ?? EMPTY_STRING_ARRAY,
      getTerminalActiveTab: (projectId) => get().terminalActiveTab[projectId] ?? 0,
      getActiveIdeChatThreadId: (projectId) => get().activeIdeChatThreadId[projectId] ?? null,
    }),
    {
      name: "xanom-ide-store",
      partialize: (state) => ({
        fileTreeWidth: state.fileTreeWidth,
        chatPanelWidth: state.chatPanelWidth,
        terminalHeight: state.terminalHeight,
        fileTreeCollapsed: state.fileTreeCollapsed,
        chatPanelCollapsed: state.chatPanelCollapsed,
        terminalVisible: state.terminalVisible,
        terminalSessions: state.terminalSessions,
        terminalActiveTab: state.terminalActiveTab,
        activeIdeChatThreadId: state.activeIdeChatThreadId,
      }),
    }
  )
);
```

```typescript
// src/components/layout/IdeLayout.tsx
import { useCallback, useRef } from "react";
import { useIdeStore } from "../../stores/ideStore";
import { useProjectStore } from "../../stores/projectStore";
import { FileTree } from "../editor/FileTree";
import { EditorTabs } from "../editor/EditorTabs";
import { CodeEditor } from "../editor/CodeEditor";
import { ResizeHandle } from "./ResizeHandle";
import { IdeTerminalPanel } from "./IdeTerminalPanel";
import { IdeChatPanel } from "./IdeChatPanel";
import { IdeToolbar } from "./IdeToolbar";
import { useEditorStore } from "../../stores/editorStore";
import { useIdeKeyboardShortcuts } from "../../hooks/useIdeKeyboardShortcuts";

const FILE_TREE_MIN = 180;
const FILE_TREE_MAX = 500;
const CHAT_MIN = 280;
const CHAT_MAX = 600;

export function IdeLayout() {
  const selectedProject = useProjectStore((s) => s.selectedProject);
  const activeTabPath = useEditorStore((s) => s.activeTabPath);

  const fileTreeWidth = useIdeStore((s) => s.fileTreeWidth);
  const chatPanelWidth = useIdeStore((s) => s.chatPanelWidth);
  const terminalHeight = useIdeStore((s) => s.terminalHeight);
  const fileTreeCollapsed = useIdeStore((s) => s.fileTreeCollapsed);
  const chatPanelCollapsed = useIdeStore((s) => s.chatPanelCollapsed);
  const terminalVisible = useIdeStore((s) => s.terminalVisible);

  const setFileTreeWidth = useIdeStore((s) => s.setFileTreeWidth);
  const setChatPanelWidth = useIdeStore((s) => s.setChatPanelWidth);
  const setTerminalHeight = useIdeStore((s) => s.setTerminalHeight);

  const centerRef = useRef<HTMLDivElement>(null);

  useIdeKeyboardShortcuts();

  const handleFileTreeResize = useCallback(
    (delta: number) => {
      setFileTreeWidth(fileTreeWidth + delta);
    },
    [fileTreeWidth, setFileTreeWidth]
  );

  const handleChatResize = useCallback(
    (delta: number) => {
      setChatPanelWidth(chatPanelWidth - delta);
    },
    [chatPanelWidth, setChatPanelWidth]
  );

  const handleTerminalResize = useCallback(
    (delta: number) => {
      if (!centerRef.current) return;
      const centerHeight = centerRef.current.clientHeight;
      const currentPx = terminalHeight * centerHeight;
      const newPx = currentPx - delta;
      setTerminalHeight(newPx / centerHeight);
    },
    [terminalHeight, setTerminalHeight]
  );

  if (!selectedProject) {
    return (
      <div className="flex h-full w-full items-center justify-center bg-zinc-950 text-zinc-400">
        <div className="text-center">
          <p className="text-lg font-medium">No project selected</p>
          <p className="mt-1 text-sm">Select a project to use IDE mode.</p>
        </div>
      </div>
    );
  }

  const projectId = selectedProject.id;
  const workDir = selectedProject.repo_path ?? selectedProject.work_dir ?? "";
  const effectiveFileTreeWidth = fileTreeCollapsed ? 0 : fileTreeWidth;
  const effectiveChatWidth = chatPanelCollapsed ? 0 : chatPanelWidth;

  return (
    <div className="flex h-screen w-screen overflow-hidden bg-zinc-950">
      {/* File Tree (Left) */}
      {!fileTreeCollapsed && (
        <div
          className="shrink-0 overflow-hidden border-r border-zinc-800"
          style={{ width: effectiveFileTreeWidth }}
        >
          <FileTree rootPath={workDir} threadId={null} />
        </div>
      )}

      {!fileTreeCollapsed && (
        <ResizeHandle direction="horizontal" onResize={handleFileTreeResize} />
      )}

      {/* Center: Editor + Terminal */}
      <div ref={centerRef} className="flex min-w-[300px] flex-1 flex-col overflow-hidden">
        {/* Editor Area */}
        <div
          className="flex flex-1 flex-col overflow-hidden"
          style={terminalVisible ? { flex: `0 0 ${(1 - terminalHeight) * 100}%` } : undefined}
        >
          <EditorTabs />
          <div className="flex-1 overflow-hidden">
            <CodeEditor filePath={activeTabPath} />
          </div>
        </div>

        {/* Terminal (Bottom) */}
        {terminalVisible && (
          <>
            <ResizeHandle direction="vertical" onResize={handleTerminalResize} />
            <div style={{ flex: `0 0 ${terminalHeight * 100}%`, minHeight: 100 }}>
              <IdeTerminalPanel projectId={projectId} workDir={workDir} />
            </div>
          </>
        )}

        {/* Toolbar */}
        <IdeToolbar />
      </div>

      {/* Chat Panel (Right) */}
      {!chatPanelCollapsed && (
        <ResizeHandle direction="horizontal" onResize={handleChatResize} />
      )}

      {!chatPanelCollapsed && (
        <div
          className="shrink-0 overflow-hidden border-l border-zinc-800"
          style={{ width: effectiveChatWidth }}
        >
          <IdeChatPanel projectId={projectId} workDir={workDir} />
        </div>
      )}
    </div>
  );
}
```

```typescript
// src/hooks/useIdeKeyboardShortcuts.ts
import { useEffect } from "react";
import { useIdeStore } from "../stores/ideStore";
import { useUiStore } from "../stores/uiStore";

export function useIdeKeyboardShortcuts() {
  const toggleFileTree = useIdeStore((s) => s.toggleFileTree);
  const toggleChatPanel = useIdeStore((s) => s.toggleChatPanel);
  const toggleTerminal = useIdeStore((s) => s.toggleTerminal);
  const appMode = useUiStore((s) => s.appMode);

  useEffect(() => {
    if (appMode !== "ide") return;

    function handleKeyDown(e: KeyboardEvent) {
      const meta = e.metaKey || e.ctrlKey;

      // Cmd+Shift+. → toggle mode (handled in App.tsx)
      // Cmd+B → toggle file tree
      if (meta && !e.shiftKey && e.key === "b") {
        e.preventDefault();
        toggleFileTree();
        return;
      }
      // Cmd+Shift+B → toggle chat panel
      if (meta && e.shiftKey && e.key === "B") {
        e.preventDefault();
        toggleChatPanel();
        return;
      }
      // Cmd+` → toggle terminal
      if (meta && e.key === "`") {
        e.preventDefault();
        toggleTerminal();
        return;
      }
    }

    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [appMode, toggleFileTree, toggleChatPanel, toggleTerminal]);
}
```

```typescript
// src/components/layout/IdeChatPanel.tsx
import { useState, useCallback, useMemo } from "react";
import { useIdeStore } from "../../stores/ideStore";
import { useThreadStore } from "../../stores/threadStore";
import { ClaudeSdkSessionView } from "../thread/ClaudeSdkSessionView";
import { DraftChatView } from "../thread/DraftChatView";
import { MessageSquarePlus, History, ChevronDown } from "lucide-react";

interface Props {
  projectId: string;
  workDir: string;
}

export function IdeChatPanel({ projectId, workDir }: Props) {
  const activeThreadId = useIdeStore((s) => s.getActiveIdeChatThreadId(projectId));
  const setActiveThread = useIdeStore((s) => s.setActiveIdeChatThread);
  const threads = useThreadStore((s) => s.threads);
  const createThread = useThreadStore((s) => s.createThread);

  const [showHistory, setShowHistory] = useState(false);

  // Filter SDK threads for this project
  const projectSdkThreads = useMemo(
    () =>
      threads.filter(
        (t) =>
          t.project_id === projectId &&
          t.interaction_mode === "sdk" &&
          !t.is_archived
      ),
    [threads, projectId]
  );

  const activeThread = useMemo(
    () => projectSdkThreads.find((t) => t.id === activeThreadId) ?? null,
    [projectSdkThreads, activeThreadId]
  );

  const handleNewChat = useCallback(async () => {
    try {
      const thread = await createThread({
        projectId,
        provider: "ClaudeCode",
        interactionMode: "sdk",
        workDir,
      });
      if (thread) {
        setActiveThread(projectId, thread.id);
      }
    } catch (err) {
      console.error("Failed to create IDE chat thread:", err);
    }
  }, [projectId, workDir, createThread, setActiveThread]);

  const handleSelectThread = useCallback(
    (threadId: string) => {
      setActiveThread(projectId, threadId);
      setShowHistory(false);
    },
    [projectId, setActiveThread]
  );

  return (
    <div className="flex h-full flex-col bg-zinc-950">
      {/* Header */}
      <div className="flex h-10 shrink-0 items-center justify-between border-b border-zinc-800 px-3">
        <div className="relative">
          <button
            onClick={() => setShowHistory(!showHistory)}
            className="flex items-center gap-1 rounded px-2 py-1 text-xs font-medium text-zinc-300 hover:bg-zinc-800"
          >
            <History size={12} />
            <span>Chat History</span>
            <ChevronDown size={10} />
          </button>

          {/* History Dropdown */}
          {showHistory && (
            <div className="absolute left-0 top-full z-50 mt-1 w-64 rounded-md border border-zinc-700 bg-zinc-900 py-1 shadow-xl">
              {projectSdkThreads.length === 0 ? (
                <div className="px-3 py-2 text-xs text-zinc-500">No chat history</div>
              ) : (
                projectSdkThreads.slice(0, 20).map((t) => (
                  <button
                    key={t.id}
                    onClick={() => handleSelectThread(t.id)}
                    className={`w-full px-3 py-1.5 text-left text-xs hover:bg-zinc-800 ${
                      t.id === activeThreadId ? "bg-zinc-800 text-zinc-100" : "text-zinc-400"
                    }`}
                  >
                    {t.name || "Untitled chat"}
```

```rust
// src-tauri/src/commands/git.rs
use std::collections::HashMap;
use std::process::Command;

#[tauri::command]
pub async fn get_git_status(work_dir: String) -> Result<HashMap<String, String>, String> {
    let output = Command::new("git")
        .args(["status", "--porcelain", "-uall"])
        .current_dir(&work_dir)
        .output()
        .map_err(|e| format!("Failed to run git status: {}", e))?;

    if !output.status.success() {
        let stderr = String::from_utf8_lossy(&output.stderr);
        return Err(format!("git status failed: {}", stderr));
    }

    let stdout = String::from_utf8_lossy(&output.stdout);
    let mut status_map: HashMap<String, String> = HashMap::new();

    for line in stdout.lines() {
        if line.len() < 4 {
            continue;
        }
        let status_code = line[..2].trim().to_string();
        let file_path = line[3..].to_string();

        // Handle renames: "R  old -> new"
        let actual_path = if let Some(arrow_idx) = file_path.find(" -> ") {
            file_path[arrow_idx + 4..].to_string()
        } else {
            file_path
        };

        status_map.insert(actual_path, status_code);
    }

    Ok(status_map)
}
```

```typescript
// src/hooks/useGitStatus.ts
import { useEffect, useRef, useState } from "react";
import { getGitStatus } from "../lib/commands";

const POLL_INTERVAL_MS = 3000;
const EMPTY_STATUS: Record<string, string> = {};

export function useGitStatus(workDir: string | null): Record<string, string> {
  const [status, setStatus] = useState<Record<string, string>>(EMPTY_STATUS);
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  useEffect(() => {
    if (!workDir) {
      setStatus(EMPTY_STATUS);
      return;
    }

    let cancelled = false;

    async function fetchStatus() {
      try {
        const result = await getGitStatus(workDir!);
        if (!cancelled) {
          setStatus(result);
        }
      } catch {
        // Not a git repo or git not available — silently ignore
        if (!cancelled) {
          setStatus(EMPTY_STATUS);
        }
      }
    }

    // Initial fetch
    fetchStatus();

    // Poll
    intervalRef.current = setInterval(fetchStatus, POLL_INTERVAL_MS);

    return () => {
      cancelled = true;
      if (intervalRef.current) {
        clearInterval(intervalRef.current);
      }
    };
  }, [workDir]);

  return status;
}
```

```typescript
import { useEditorStore } from "../stores/editorStore";

const FILE_MUTATING_TOOLS = new Set(["Edit", "Write", "ApplyPatch", "NotebookEdit"]);
const FILE_READING_TOOLS = new Set(["Read"]);

interface ToolStartPayload {
  type: "tool.started";
  toolUseId: string;
  toolName: string;
  input: string; // JSON string of tool input
}

interface ToolCompletedPayload {
  type: "tool.completed";
  toolUseId: string;
  content: string;
  isError: boolean;
}

type SdkEventPayload = ToolStartPayload | ToolCompletedPayload | { type: string };

export function useAutoOpenOnAiEdit(sessionId: string | null) {
  const openTab = useEditorStore((s) => s.openTab);
  const markAiEdited = useEditorStore((s) => s.markAiEdited);
  const refreshFileFromDisk = useEditorStore((s) => s.refreshFileFromDisk);

    if (!sessionId) return;

        if (cancelled) return;

        if (data.type === "tool.started") {

            if (isDirty) {
              // TODO: Show conflict dialog in v1 — for now, just refresh
              console.warn(`AI edited dirty file: ${filePath}`);
            }

            // Refresh content from disk + open tab
            refreshFileFromDisk(filePath);
            openTab(filePath);
            markAiEdited(filePath);

            // Clear AI-edited badge after 5 seconds
            setTimeout(() => {
              useEditorStore.getState().clearAiEdited(filePath);
            }, 5000);
          } else if (FILE_READING_TOOLS.has(toolName)) {
            openTab(filePath);
          }
        }
      });
    }

    setup();
    return () => {
      cancelled = true;
      unlisten?.();
      toolStartsRef.current.clear();
    };
  }, [sessionId, openTab, markAiEdited, refreshFileFromDisk]);
```

```typescript
// src/lib/fuzzyMatch.ts

export interface FuzzyResult {
  item: string;
  score: number;
}

/**
 * Simple fuzzy match scorer.
 * Returns a score > 0 if query fuzzy-matches target, 0 if no match.
 * Higher score = better match.
 */
export function fuzzyScore(query: string, target: string): number {
  const q = query.toLowerCase();
  const t = target.toLowerCase();

  if (q.length === 0) return 1;
  if (q.length > t.length) return 0;

  let score = 0;
  let qi = 0;
  let consecutiveBonus = 0;

  for (let ti = 0; ti < t.length && qi < q.length; ti++) {
    if (t[ti] === q[qi]) {
      score += 1 + consecutiveBonus;
      consecutiveBonus += 1;
      qi++;

      // Bonus for matching after separator
      if (ti === 0 || t[ti - 1] === "/" || t[ti - 1] === "\\") {
        score += 3;
      }
    } else {
      consecutiveBonus = 0;
    }
  }

  // All query chars must match
  if (qi < q.length) return 0;

  // Bonus for shorter targets (more specific match)
  score += Math.max(0, 20 - t.length) * 0.1;

  return score;
}

export function fuzzyFilter(query: string, items: string[], limit = 50): FuzzyResult[] {
  if (!query) return items.slice(0, limit).map((item) => ({ item, score: 1 }));

  const results: FuzzyResult[] = [];
  for (const item of items) {
    const score = fuzzyScore(query, item);
    if (score > 0) {
      results.push({ item, score });
    }
  }

  results.sort((a, b) => b.score - a.score);
  return results.slice(0, limit);
}
```

```typescript
// src/components/layout/QuickOpenDialog.tsx
import { useState, useEffect, useCallback, useRef, useMemo } from "react";
import { fuzzyFilter } from "../../lib/fuzzyMatch";
import { useEditorStore } from "../../stores/editorStore";
import { listDirectoryEntries } from "../../lib/commands";
import { FileIcon } from "../editor/FileIcon";
import { Search } from "lucide-react";

interface Props {
  workDir: string;
  open: boolean;
  onClose: () => void;
}

export function QuickOpenDialog({ workDir, open, onClose }: Props) {
  const [query, setQuery] = useState("");
  const [allFiles, setAllFiles] = useState<string[]>([]);
  const [selectedIndex, setSelectedIndex] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);
  const openTab = useEditorStore((s) => s.openTab);

  // Load file list on open
  useEffect(() => {
    if (!open) {
      setQuery("");
      setSelectedIndex(0);
      return;
    }
    inputRef.current?.focus();

    async function loadFiles() {
      try {
        const entries = await listDirectoryEntries(workDir, "", false);
        // Flatten recursively — collect file paths only
        const files = entries
          .filter((e: { isDirectory: boolean }) => !e.isDirectory)
          .map((e: { relativePath: string }) => e.relativePath);
        setAllFiles(files);
      } catch (err) {
        console.error("QuickOpen: failed to load files", err);
      }
    }
    loadFiles();
  }, [open, workDir]);

  const filtered = useMemo(() => fuzzyFilter(query, allFiles, 50), [query, allFiles]);

  const handleSelect = useCallback(
    (relativePath: string) => {
      const fullPath = `${workDir}/${relativePath}`;
      openTab(fullPath);
      onClose();
    },
    [workDir, openTab, onClose]
  );

  const handleKeyDown = useCallback(
    (e: React.KeyboardEvent) => {
      if (e.key === "Escape") {
        onClose();
        return;
      }
      if (e.key === "ArrowDown") {
        e.preventDefault();
        setSelectedIndex((i) => Math.min(i + 1, filtered.length - 1));
        return;
      }
      if (e.key === "ArrowUp") {
        e.preventDefault();
        setSelectedIndex((i) => Math.max(i - 1, 0));
        return;
      }
      if (e.key === "Enter" && filtered[selectedIndex]) {
        handleSelect(filtered[selectedIndex].item);
      }
    },
    [filtered, selectedIndex, handleSelect, onClose]
  );

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-[200] flex items-start justify-center pt-[15vh]" onClick={onClose}>
      <div
        className="w-[520px] rounded-lg border border-zinc-700 bg-zinc-900 shadow-2xl"
        onClick={(e) => e.stopPropagation()}
      >
        {/* Search Input */}
        <div className="flex items-center gap-2 border-b border-zinc-800 px-3 py-2">
          <Search size={14} className="text-zinc-500" />
          <input
            ref={inputRef}
            type="text"
            value={query}
            onChange={(e) => {
              setQuery(e.target.value);
              setSelectedIndex(0);
            }}
            onKeyDown={handleKeyDown}
            className="flex-1 bg-transparent text-sm text-zinc-100 outline-none placeholder:text-zinc-500"
            placeholder="Search files by name..."
          />
        </div>

        {/* Results */}
        <div className="max-h-[300px] overflow-y-auto py-1">
          {filtered.length === 0 ? (
            <div className="px-3 py-4 text-center text-xs text-zinc-500">
              {query ? "No matching files" : "Loading..."}
            </div>
          ) : (
            filtered.map((result, idx) => (
              <button
                key={result.item}
                onClick={() => handleSelect(result.item)}
                className={`flex w-full items-center gap-2 px-3 py-1.5 text-left text-xs ${
                  idx === selectedIndex
                    ? "bg-blue-600/20 text-zinc-100"
                    : "text-zinc-400 hover:bg-zinc-800"
                }`}
              >
                <FileIcon name={result.item.split("/").pop() ?? ""} size={14} />
                <span className="truncate">{result.item}</span>
              </button>
            ))
          )}
        </div>
      </div>
    </div>
  );
}
```
